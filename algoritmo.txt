CONSTANTES:
    MAX_POKEMON = 1025
    TAM_NOMBRE = 50
    TAM_TIPO = 20
    TAM_LINEA = 256

ESTRUCTURA Pokemon:
    entero iId
    cadena cNombre[TAM_NOMBRE]
    cadena cTipo1[TAM_TIPO]
    cadena cTipo2[TAM_TIPO]
    entero iEncontrado (0 = No, 1 = Sí)

ESTRUCTURA tipoNodo (Nodo de la Pila):
    Pokemon infoPokemon
    apuntador tipoNodo pSiguiente

TIPO pNodo = apuntador a tipoNodo

VARIABLES GLOBALES:
    Pokemon arregloPokedex[MAX_POKEMON]
    entero iTotalPokemones = 0
    pNodo tope_pila = NULL
    entero iCapturados = 0

FUNCION entero pila_vacia(pNodo pila)
    SI pila == NULL ENTONCES Retornar 1
    SINO Retornar 0
FIN FUNCION

FUNCION void Push(pNodo *ultimo, Pokemon nuevoPokemon)
    pNodo nuevoNodo = (pNodo) malloc(sizeof(tipoNodo))
    nuevoNodo->infoPokemon = nuevoPokemon
    nuevoNodo->pSiguiente = *ultimo
    *ultimo = nuevoNodo
FIN FUNCION

FUNCION void Pop(pNodo *pila, Pokemon *pokemonSacado)
    pNodo nodoAux = *pila
    *pokemonSacado = nodoAux->infoPokemon
    *pila = nodoAux->pSiguiente
    free(nodoAux)
FIN FUNCION

    Retornar -1

FUNCION void capturar_pokemon()
    cadena cNombreAux[50]
    entero iIndex = -1
    
    Imprimir "Ingrese el nombre o ID del Pokémon a capturar: "
    Leer cNombreAux
    
    SI cNombreAux[0] >= '0' Y cNombreAux[0] <= '9' ENTONCES
        // Es un número, convertir a entero y buscar por ID
        iIndex = buscar_por_id(atoi(cNombreAux))
    SINO
        iIndex = buscar_por_nombre(cNombreAux)
    FIN SI
    
    SI iIndex == -1 ENTONCES
        Imprimir "Pokémon no encontrado"
    SINO
        SI arregloPokedex[iIndex].iEncontrado == 1 ENTONCES
            Imprimir "Ya tienes este Pokémon"
        SINO
            arregloPokedex[iIndex].iEncontrado = 1
            iCapturados = iCapturados + 1
            Push(&tope_pila, arregloPokedex[iIndex])
            Imprimir "¡Pokémon capturado con éxito!"
        FIN SI
    FIN SI
FIN FUNCION
